## PART B: NVIDIA PROVIDER CREATION

### Agent: Codex
### Duration: 20 minutes

### Prompt
```
TASK: Create NVIDIA provider for Ultra-Dex

CREATE FILE: apps/cli/lib/providers/nvidia.js

CONTENT:
```javascript
/**
 * NVIDIA Provider for Ultra-Dex
 * Uses multi-key rotation system from nemotron.js
 */

import { logger } from '../utils/logger.js';

// Lazy load to avoid circular dependencies
let _nvidiaModule = null;

async function getNVIDIAModule() {
  if (!_nvidiaModule) {
    try {
      _nvidiaModule = await import('../../../src/services/ai-providers/nemotron.js');
    } catch (error) {
      // Fallback if nemotron.js doesn't exist
      _nvidiaModule = {
        initNVIDIAKeys: () => {},
        createRotatingClient: (model) => ({
          client: {
            chat: {
              completions: {
                create: async () => ({ choices: [{ message: { content: 'Mock response' } }] })
              }
            }
          },
          keyIndex: 0
        })
      };
    }
  }
  return _nvidiaModule;
}

export class NVIDIAProvider {
  constructor(apiKey, options = {}) {
    this.model = options.model || "nvidia/nemotron-3-super-120b-a12b";
    this._initialized = false;
    this.apiKey = apiKey || process.env.NVIDIA_API_KEY;
  }

  async _ensureInit() {
    if (!this._initialized) {
      const mod = await getNVIDIAModule();
      mod.initNVIDIAKeys?.();
      this._initialized = true;
    }
  }

  async generate(systemPrompt, userPrompt, options = {}) {
    await this._ensureInit();
    
    const model = options.model || this.model;
    const mod = await getNVIDIAModule();
    const { client, keyIndex } = mod.createRotatingClient(model);
    
    const run_id = options.run_id || 'unknown';
    
    logger.debug?.("NVIDIA request", { model, keyIndex, run_id });
    
    const startTime = Date.now();
    
    try {
      const messages = [
        { role: "system", content: systemPrompt },
        { role: "user", content: userPrompt }
      ];
      
      const res = await client.chat.completions.create({
        model,
        messages,
        temperature: options.temperature || 0.7,
        max_tokens: options.max_tokens || 4096
      });
      
      const duration = Date.now() - startTime;
      
      logger.debug?.("NVIDIA response", { model, duration_ms: duration, run_id });
      
      return {
        content: res.choices[0].message.content,
        model,
        usage: res.usage,
        duration_ms: duration
      };
    } catch (error) {
      logger.error?.("NVIDIA error", { model, error: error.message, run_id });
      throw error;
    }
  }

  // Alternative method signature for compatibility
  async chat(messages, options = {}) {
    await this._ensureInit();
    
    const model = options.model || this.model;
    const mod = await getNVIDIAModule();
    const { client } = mod.createRotatingClient(model);
    
    const res = await client.chat.completions.create({
      model,
      messages,
      temperature: options.temperature || 0.7,
      max_tokens: options.max_tokens || 4096
    });
    
    return res.choices[0].message.content;
  }

  getName() {
    return "nvidia";
  }

  getModel() {
    return this.model;
  }
}

export default NVIDIAProvider;
```

CONSTRAINT:

- NO extra features
- Include fallback for missing nemotron.js
