```javascript
import fs from 'fs';
import path from 'path';

export class ExecutionTrace {
  constructor(run_id) {
    this.run_id = run_id;
    this.started_at = new Date().toISOString();
    this.steps = [];
    this.status = 'running';
  }

  addStep({ step, agent, action, input, output, status, duration_ms }) {
    this.steps.push({
      step,
      agent,
      action,
      input: typeof input === 'string' ? input.slice(0, 500) : JSON.stringify(input).slice(0, 500),
      output: typeof output === 'string' ? output.slice(0, 500) : JSON.stringify(output).slice(0, 500),
      status,
      duration_ms,
      timestamp: new Date().toISOString()
    });
  }

  complete(status = 'success') {
    this.completed_at = new Date().toISOString();
    this.status = status;
    this.total_steps = this.steps.length;
    this.duration_ms = new Date(this.completed_at) - new Date(this.started_at);
  }

  async save() {
    const dir = path.join(process.cwd(), '.ultra', 'traces');
    await fs.promises.mkdir(dir, { recursive: true });
    const file = path.join(dir, `${this.run_id}.json`);
    await fs.promises.writeFile(file, JSON.stringify(this, null, 2));
    return file;
  }

  toSummary() {
    return `
=== TRACE SUMMARY ===
run_id: ${this.run_id}
steps: ${this.steps.length}
status: ${this.status}
duration: ${this.duration_ms}ms
    `.trim();
  }
}
```

import { initNVIDIAKeys, createRotatingClient } from '../../../src/services/ai-providers/nemotron.js';
import { logger } from '../utils/logger.js';

export class NVIDIAProvider {
  constructor(options = {}) {
    this.defaultModel = options.model || 'nvidia/nemotron-3-super-120b-a12b';
    this._initialized = false;
  }

  async _ensureInit() {
    if (!this._initialized) {
      initNVIDIAKeys();
      this._initialized = true;
    }
  }

  async generate(messages, options = {}) {
    await this._ensureInit();
    
    const model = options.model || this.defaultModel;
    const { client, keyIndex } = createRotatingClient(model);
    
    logger.debug('NVIDIA request', { 
      model, 
      keyIndex, 
      messageCount: messages.length,
      run_id: options.run_id 
    });
    
    const startTime = Date.now();
    
    try {
      const response = await client.chat.completions.create({
        model,
        messages,
        temperature: options.temperature || 0.7,
        max_tokens: options.max_tokens || 4096
      });
      
      const duration = Date.now() - startTime;
      
      logger.debug('NVIDIA response', { 
        model, 
        duration_ms: duration,
        run_id: options.run_id 
      });
      
      return {
        content: response.choices[0].message.content,
        model,
        usage: response.usage,
        duration_ms: duration
      };
    } catch (error) {
      logger.error('NVIDIA error', { 
        model, 
        error: error.message,
        run_id: options.run_id 
      });
      throw error;
    }
  }
}
