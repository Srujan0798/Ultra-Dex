Issues:
- No strict step control
- No deterministic DAG
- No bounded execution
- Will break at scale

## TARGET STATE

```
BEFORE:
while (true) {  // ❌ DANGEROUS
  response = await model.call();

AFTER:
const MAX_STEPS = 10;  // ✅ BOUNDED
let step = 0;
while (step < MAX_STEPS && !isComplete) {
  step++;
  response = await model.call();
  if (isDone(response)) break;
}
```

```javascript
// Configuration
const DEFAULT_MAX_STEPS = 10;

// In run function or agent loop
async function runAgentLoop({ goal, provider, maxSteps = DEFAULT_MAX_STEPS, run_id, ...options }) {
  const logger = options.logger || console;
  let currentStep = 0;
  let isComplete = false;
  let lastResult = null;

  logger.info?.("Agent loop started", { run_id, goal, maxSteps });

  while (currentStep < maxSteps && !isComplete) {
    currentStep++;
    const stepStart = Date.now();

    try {
      // Execute step
      logger.debug?.("Step started", { run_id, step: currentStep });
      
      const result = await executeStep({
        step: currentStep,
        goal,
        provider,
        previousResult: lastResult,
        run_id
      });

      lastResult = result;

      // Check completion
      isComplete = isGoalComplete(result);
      
      const duration = Date.now() - stepStart;
      logger.debug?.("Step completed", { run_id, step: currentStep, duration_ms: duration, isComplete });

      // Record in trace
      if (options.trace) {
        options.trace.addStep({
          step: currentStep,
          agent: options.agent || 'executor',
          action: result.action || 'EXECUTE',
          input: result.input?.slice?.(0, 200) || '',
          output: result.output?.slice?.(0, 200) || '',
          status: 'success',
          duration_ms: duration
        });
      }

    } catch (error) {
      logger.error?.("Step failed", { run_id, step: currentStep, error: error.message });
      
      if (options.trace) {
        options.trace.addStep({
          step: currentStep,
          agent: options.agent || 'executor',
          action: 'ERROR',
          input: '',
          output: error.message,
          status: 'error',
          duration_ms: Date.now() - stepStart
        });
      }

      throw error;
    }
  }

  // Check if hit max steps
  if (currentStep >= maxSteps && !isComplete) {
    logger.warn?.("Max steps reached without completion", { 
      run_id, 
      maxSteps, 
      step: currentStep 
    });
    
    return {
      status: 'max_steps_reached',
      steps: currentStep,
      result: lastResult,
      message: `Execution stopped after ${maxSteps} steps without completion`
    };
  }

  logger.info?.("Agent loop completed", { run_id, steps: currentStep });

  return {
    status: 'completed',
    steps: currentStep,
    result: lastResult
  };
}

// Helper: Check if goal is complete
function isGoalComplete(result) {
  if (!result) return false;
  
  // Check for explicit completion signals
  if (result.isComplete === true) return true;
  if (result.status === 'complete') return true;
  if (result.status === 'done') return true;
  
  // Check for final output
  if (result.finalOutput) return true;
  
  return false;
}
```

## BOUNDED LOOP PATTERN

```javascript
// ❌ WRONG - Unbounded
while (true) {
  result = await step();
  if (result.done) break;  // Could run forever
}

// ❌ WRONG - No control
for (const step of steps) {
  await execute(step);  // Could be infinite steps
}

// ✅ CORRECT - Bounded
const MAX = 10;
let i = 0;
while (i < MAX && !done) {
  i++;
  result = await step();
  done = checkDone(result);
}
if (i >= MAX) warnMaxReached();
```
