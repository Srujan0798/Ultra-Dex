## L0 — EXECUTION ENGINE (Kernel)

### Purpose
Deterministic execution of tasks.

### Structure
```javascript
class ExecutionTask {
  constructor() {
    this.id = generateId();
    this.input = null;
    this.agent = null;
    this.steps = [];
    this.status = 'pending';
    this.trace = null;
  }
}
```

### Engine
```javascript
class ExecutionEngine {
  constructor(options = {}) {
    this.maxSteps = options.maxSteps || 10;
    this.trace = new ExecutionTrace(generateRunId());
  }

  async run(task) {
    task.status = 'running';
    
    for (let i = 0; i < task.steps.length && i < this.maxSteps; i++) {
      const step = task.steps[i];
      const startTime = Date.now();
      
      try {
        const result = await this.executeStep(step);
        
        this.trace.addStep({
          step: i + 1,
          agent: step.agent,
          action: step.action,
          input: step.input,
          output: result,
          status: 'success',
          duration_ms: Date.now() - startTime
        });
        
        step.result = result;
        
      } catch (error) {
        this.trace.addStep({
          step: i + 1,
          agent: step.agent,
          action: step.action,
          status: 'error',
          output: error.message,
          duration_ms: Date.now() - startTime
        });
        
        task.status = 'failed';
        throw error;
      }
    }
    
    task.status = 'completed';
    this.trace.complete('success');
    return task;
  }

  async executeStep(step) {
    const agent = CapabilityRouter.match(step);
    if (!agent) {
      throw new Error(`No agent capable of: ${step.action}`);
    }
    return agent.execute(step);
  }
}
```

### Rules
- No randomness
- No free agent decisions
- Everything step-driven
- Bounded execution (max_steps)

### 2. Scheduler (deterministic)
```javascript
class Scheduler {
  constructor(capabilityRouter) {
    this.router = capabilityRouter;
    this.queue = [];
  }

  async schedule(steps) {
    const scheduledSteps = [];
    
    for (const step of steps) {
      const agent = this.router.match(step);
      
      if (!agent) {
        throw new Error(`Cannot schedule step: ${step.action} - no capable agent`);
      }
      
      scheduledSteps.push({
        ...step,
        assignedAgent: agent.name,
        priority: this.calculatePriority(step)
      });
    }
    
    // Sort by priority (dependencies first)
    return scheduledSteps.sort((a, b) => b.priority - a.priority);
  }

  calculatePriority(step) {
    // Higher priority for foundational tasks
    const priorities = {
      'design': 100,
      'create': 90,
      'implement': 80,
      'test': 70,
      'document': 60
    };
    
    for (const [keyword, priority] of Object.entries(priorities)) {
      if (step.action.toLowerCase().includes(keyword)) {
        return priority;
      }
    }
    return 50;
  }
}
```

## L2 — CAPABILITY ROUTER (Critical New Component)

### This is your MISSING piece

### Structure
```javascript
class CapabilityRouter {
  constructor() {
    this.agents = new Map();
  }

  register(agent) {
    this.agents.set(agent.name, {
      name: agent.name,
      capabilities: agent.capabilities,
      skills: agent.skills,
      limitations: agent.limitations
    });
  }

  match(task) {
    const required = this.extractRequirements(task);
    
    for (const [name, agent] of this.agents) {
      const matches = required.every(req => 
        agent.skills.includes(req) || 
        agent.capabilities.includes(req)
      );
      
      if (matches) {
        return agent;
      }
    }
    
    return null; // NO MATCH → NO EXECUTION
  }

  extractRequirements(task) {
    // Extract required skills from task
    const skills = [];
    
    if (task.action.includes('API') || task.action.includes('backend')) {
      skills.push('nodejs', 'api');
    }
    if (task.action.includes('database') || task.action.includes('schema')) {
      skills.push('database', 'sql');
    }
    if (task.action.includes('test')) {
      skills.push('testing');
    }
    
    return skills;
  }
}
```

### Usage Example
```javascript
const router = new CapabilityRouter();

// Register agents
router.register({
  name: 'backend_agent',
  capabilities: ['implementation', 'api_design'],
  skills: ['nodejs', 'api', 'express', 'database'],
  limitations: ['no_frontend']
});

router.register({
  name: 'frontend_agent',
  capabilities: ['ui', 'components'],
  skills: ['react', 'css', 'typescript'],
  limitations: ['no_backend']
});

// Match task
const task = { action: 'create API endpoint', required: ['nodejs', 'api'] };
const agent = router.match(task);
// Returns: backend_agent
```

```javascript
class Memory {
  constructor(run_id) {
    this.run_id = run_id;
    this.shortTerm = new Map();  // Current task context
    this.longTerm = null;        // Git-based (.ai/ folder)
  }

  // Short-term: current execution
  setContext(key, value) {
    this.shortTerm.set(key, value);
  }

  getContext(key) {
    return this.shortTerm.get(key);
  }

  // Long-term: persisted
  async persist() {
    const data = Object.fromEntries(this.shortTerm);
    await fs.writeFile(
      `.ai/memory/${this.run_id}.json`,
      JSON.stringify(data, null, 2)
    );
  }

  async load(run_id) {
    const data = await fs.readFile(`.ai/memory/${run_id}.json`, 'utf-8');
    return JSON.parse(data);
  }
}
```

## L5 — OBSERVABILITY (Mandatory)

### ExecutionTrace Structure
```javascript
class ExecutionTrace {
  constructor(run_id) {
    this.run_id = run_id;
    this.started_at = new Date().toISOString();
    this.steps = [];
    this.agents = [];
    this.duration_ms = null;
    this.status = 'running';
  }

  addStep({ step, agent, action, input, output, status, duration_ms }) {
    this.steps.push({
      step,
      agent,
      action,
      input: truncate(input, 500),
      output: truncate(output, 500),
      status,
      duration_ms,
      timestamp: new Date().toISOString()
    });
    
    if (!this.agents.includes(agent)) {
      this.agents.push(agent);
    }
  }

  complete(status = 'success') {
    this.completed_at = new Date().toISOString();
    this.status = status;
    this.duration_ms = new Date(this.completed_at) - new Date(this.started_at);
  }
}
```
