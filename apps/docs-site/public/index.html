Quick Start Example

Get started with Ultra-Dex in just a few lines of code. 

// Initialize Ultra-Dex client
import { UltraDex } from '@ultra-dex/sdk';

const client = new UltraDex({
  apiKey: process.env.ULTRA_DEX_API_KEY,
  endpoint: 'https://api.ultra-dex.ai'
});

// Create an AI agent
const agent = await client.agents.create({
  name: 'data-analyst',
  role: 'Analyze data and generate insights',
  capabilities: ['data-analysis', 'visualization', 'reporting']
});

// Execute a task
const result = await client.agents.execute(agent.id, {
  task: 'Analyze the sales data and identify trends',
  data: salesData,
  format: 'detailed-report'
});

console.log(result);
