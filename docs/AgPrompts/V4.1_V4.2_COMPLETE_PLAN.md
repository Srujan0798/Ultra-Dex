export function createNotionClient(apiKey) {
  return new Client({ auth: apiKey });

export async function syncToNotion(client, databaseId, planData) {
  const pages = [];
  
  for (const section of planData.sections) {
    const page = await client.pages.create({
      parent: { database_id: databaseId },
      properties: {
        'Name': { title: [{ text: { content: section.name } }] },
        'Status': { select: { name: section.status } },
        'Priority': { select: { name: section.priority } },
        'Description': { rich_text: [{ text: { content: section.description } }] }
      }
    });
    pages.push(page);
  }
  
  return pages;
}

```javascript
export class AgentSwarm {
  constructor(agents = []) {
    this.agents = agents;
    this.results = new Map();
  }
  
  async runParallel(task) {
    const promises = this.agents.map(agent => 
      agent.execute(task).catch(err => ({ error: err.message }))
    );
    return Promise.allSettled(promises);
  }
  
  async runSequential(task) {
    const results = [];
    for (const agent of this.agents) {
      const result = await agent.execute(task);
      results.push(result);
      if (result.shouldStop) break;
    }
    return results;
  }
  
  async runWaterfall(task) {
    let context = task;
    for (const agent of this.agents) {
      context = await agent.execute(context);
    }
    return context;
  }
}
```
