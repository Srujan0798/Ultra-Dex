CONTENT:
```javascript
/**
 * ExecutionTrace - Records every step of Ultra-Dex execution
 * Enables debugging, replay, and MUNI integration
 */

import fs from 'fs';
import path from 'path';

export class ExecutionTrace {
  constructor(run_id) {
    this.run_id = run_id;
    this.started_at = new Date().toISOString();
    this.steps = [];
    this.status = 'running';
    this.metadata = {};
  }

  /**
   * Set metadata for the trace
   */
  setMetadata(key, value) {
    this.metadata[key] = value;
  }

  /**
   * Add a step to the trace
   */
  addStep({ step, agent, action, input, output, status, duration_ms }) {
    const entry = {
      step: step || this.steps.length + 1,
      agent: agent || 'unknown',
      action: action || 'UNKNOWN',
      input: this._truncate(input, 500),
      output: this._truncate(output, 500),
      status: status || 'unknown',
      duration_ms: duration_ms || 0,
      timestamp: new Date().toISOString()
    };
    
    this.steps.push(entry);
    return entry;
  }

  /**
   * Mark trace as complete
   */
  complete(status = 'success') {
    this.completed_at = new Date().toISOString();
    this.status = status;
    this.total_steps = this.steps.length;
    this.duration_ms = new Date(this.completed_at) - new Date(this.started_at);
  }

  /**
   * Mark trace as failed
   */
  fail(error) {
    this.completed_at = new Date().toISOString();
    this.status = 'failed';
    this.error = {
      message: error.message || String(error),
      stack: error.stack
    };
    this.total_steps = this.steps.length;
    this.duration_ms = new Date(this.completed_at) - new Date(this.started_at);
  }

  /**
   * Save trace to file
   */
  async save() {
    const dir = path.join(process.cwd(), '.ultra', 'traces');
    await fs.promises.mkdir(dir, { recursive: true });
    
    const filename = `${this.run_id}.json`;
    const filepath = path.join(dir, filename);
    
    await fs.promises.writeFile(
      filepath, 
      JSON.stringify(this.toJSON(), null, 2)
    );
    
    return filepath;
  }

  /**
   * Get full trace as JSON
   */
  toJSON() {
    return {
      run_id: this.run_id,
      started_at: this.started_at,
      completed_at: this.completed_at,
      status: this.status,
      total_steps: this.steps.length,
      duration_ms: this.duration_ms,
      metadata: this.metadata,
      steps: this.steps,
      error: this.error
    };
  }

  /**
   * Get human-readable summary
   */
  toSummary() {
    const status_icon = this.status === 'success' ? '✅' : 
                        this.status === 'failed' ? '❌' : '⏳';
    
    return `
═══════════════════════════════════════════
${status_icon} EXECUTION TRACE SUMMARY
═══════════════════════════════════════════
Run ID:   ${this.run_id}
Status:   ${this.status}
Steps:    ${this.steps.length}
Duration: ${this.duration_ms || 'N/A'}ms
Started:  ${this.started_at}
Ended:    ${this.completed_at || 'N/A'}
═══════════════════════════════════════════
    `.trim();
  }

  /**
   * Get step-by-step breakdown
   */
  toStepBreakdown() {
    if (this.steps.length === 0) {
      return 'No steps recorded.';
    }
    
    return this.steps.map(s => {
      const status_icon = s.status === 'success' ? '✓' : 
                          s.status === 'error' ? '✗' : '○';
      return `${status_icon} Step ${s.step}: ${s.action} (${s.duration_ms}ms)`;
    }).join('\n');
  }

  /**
   * Truncate string for storage
   */
  _truncate(value, maxLength) {
    if (value === undefined || value === null) return '';
    const str = typeof value === 'string' ? value : JSON.stringify(value);
    return str.length > maxLength ? str.slice(0, maxLength) + '...' : str;
  }
}

/**
 * Generate unique run_id
 */
export function generateRunId() {
  const timestamp = Date.now();
  const random = Math.random().toString(36).slice(2, 8);
  return `run_${timestamp}_${random}`;
}

export default ExecutionTrace;
```

OUTPUT:
- File created at apps/cli/lib/utils/trace.js
- Exports ExecutionTrace class and generateRunId function
```

### Validation
```bash
node -e "
import('./apps/cli/lib/utils/trace.js').then(m => {

  const trace = new ExecutionTrace(run_id);
  trace.addStep({ step: 1, agent: 'test', action: 'TEST', input: 'x', output: 'y', status: 'success', duration_ms: 100 });

## PART B: INTEGRATE TRACE IN RUN.JS

### Agent: Gemini CLI
### Duration: 10 minutes

### Prompt
```
TASK: Integrate ExecutionTrace into run.js

FILE: apps/cli/lib/commands/run.js

ADD AT TOP:
import { ExecutionTrace, generateRunId } from '../utils/trace.js';

ADD AT START OF EXECUTION:
const run_id = generateRunId();
const trace = new ExecutionTrace(run_id);
trace.setMetadata('goal', options.task || options.t);
trace.setMetadata('provider', options.provider || 'nvidia');

ADD AFTER EACH MAJOR STEP:
trace.addStep({
  step: currentStep,
  agent: agentName,
  action: 'MODEL_CALL', // or TOOL_EXECUTE, PLANNING, etc.
  input: prompt.slice(0, 200),
  output: response.slice(0, 200),
  status: 'success',
  duration_ms: Date.now() - stepStart
});

ADD AT END (SUCCESS):
trace.complete('success');
await trace.save();
console.log(trace.toSummary());

ADD AT END (FAILURE):
trace.fail(error);
await trace.save();
console.error(trace.toSummary());

CONSTRAINT:
- Do NOT change execution logic
- ONLY add trace recording
```

## PART C: ADD RUN_ID PROPAGATION

### Agent: Gemini CLI
### Duration: 10 minutes

### Prompt
```
TASK: Pass run_id through execution chain

REQUIREMENT:
Every function in execution path must receive run_id.

PATTERN:
```javascript
// At entry
const run_id = generateRunId();

// Pass to functions
await runAgentLoop({ run_id, goal, provider, ...options });

// In agent loop
async function runAgentLoop({ run_id, goal, provider, ...options }) {
  logger.info("Agent loop started", { run_id, goal });
  
  // Pass to provider
  const response = await provider.generate(systemPrompt, userPrompt, { run_id });
  
  logger.info("Agent loop completed", { run_id });
}

// In provider
logger.debug("Provider call", { run_id, model });
```

FILES TO MODIFY:
- apps/cli/lib/commands/run.js
- apps/cli/lib/autonomous/agent.js (if used)

RULE:
Every logger call must include run_id
```
