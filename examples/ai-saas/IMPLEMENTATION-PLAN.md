### SECTION 10: Data Model (Prisma)
```prisma
model User {
  id        String   @id @default(cuid())
  email     String   @unique
  chats     Chat[]
  usage     TokenUsage?
}

model Chat {
  id        String    @id @default(cuid())
  title     String
  userId    String
  user      User      @relation(fields: [userId], references: [id])
  messages  Message[]
}

model Message {
  id        String   @id @default(cuid())
  role      String   // user, assistant
  content   String
  chatId    String
  chat      Chat     @relation(fields: [chatId], references: [id])
}
```
